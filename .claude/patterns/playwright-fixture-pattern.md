# Pattern: Playwright Fixtures — Auth & API Mocking

## Overview

Fixtures provide reusable test setup/teardown. Never authenticate manually
inside individual test functions — always use the auth fixture.

## Auth Fixture

```typescript
// apps/web/e2e/fixtures/auth.fixture.ts
import { test as base, type Page, type BrowserContext } from '@playwright/test';
import path from 'path';

export type UserRole = 'superadmin' | 'admin' | 'viewer' | 'readonly';

export type AuthFixtures = {
  /** Authenticated page — defaults to Superadmin */
  authenticatedPage: Page;
  /** Authenticated as a specific role */
  pageAs: (role: UserRole) => Promise<Page>;
};

// Storage state paths — generated by global setup
const AUTH_STATE: Record<UserRole, string> = {
  superadmin: 'apps/web/e2e/.auth/superadmin.json',
  admin:      'apps/web/e2e/.auth/admin.json',
  viewer:     'apps/web/e2e/.auth/viewer.json',
  readonly:   'apps/web/e2e/.auth/readonly.json',
};

export const test = base.extend<AuthFixtures>({
  // Default: authenticated as superadmin
  authenticatedPage: async ({ browser }, use) => {
    const context = await browser.newContext({
      storageState: AUTH_STATE.superadmin,
    });
    const page = await context.newPage();
    await use(page);
    await context.close();
  },

  // Parameterised role
  pageAs: async ({ browser }, use) => {
    const contexts: BrowserContext[] = [];
    const factory = async (role: UserRole): Promise<Page> => {
      const context = await browser.newContext({
        storageState: AUTH_STATE[role],
      });
      contexts.push(context);
      return context.newPage();
    };
    await use(factory);
    await Promise.all(contexts.map(ctx => ctx.close()));
  },
});

export { expect } from '@playwright/test';
```

## API Mock Fixture

```typescript
// apps/web/e2e/fixtures/api-mock.fixture.ts
import { test as base, type Page, type Route } from '@playwright/test';

type MockOptions = {
  status?: number;
  body?: unknown;
  delay?: number;
  headers?: Record<string, string>;
};

type ApiMockFixture = {
  mockApi: {
    get(urlPattern: string, body: unknown, options?: MockOptions): Promise<void>;
    post(urlPattern: string, body: unknown, options?: MockOptions): Promise<void>;
    put(urlPattern: string, body: unknown, options?: MockOptions): Promise<void>;
    delete(urlPattern: string, options?: MockOptions): Promise<void>;
    error(urlPattern: string, status: number, message?: string): Promise<void>;
    networkError(urlPattern: string): Promise<void>;
    passThrough(urlPattern: string): Promise<void>;
  };
};

export const test = base.extend<ApiMockFixture>({
  mockApi: async ({ page }, use) => {
    const mock = {
      async get(urlPattern: string, body: unknown, opts: MockOptions = {}) {
        await page.route(`**${urlPattern}**`, async route => {
          if (route.request().method() !== 'GET') { await route.continue(); return; }
          if (opts.delay) await new Promise(r => setTimeout(r, opts.delay));
          await route.fulfill({
            status: opts.status ?? 200,
            contentType: 'application/json',
            body: JSON.stringify(body),
            headers: opts.headers,
          });
        });
      },

      async post(urlPattern: string, body: unknown, opts: MockOptions = {}) {
        await page.route(`**${urlPattern}**`, async route => {
          if (route.request().method() !== 'POST') { await route.continue(); return; }
          if (opts.delay) await new Promise(r => setTimeout(r, opts.delay));
          await route.fulfill({
            status: opts.status ?? 201,
            contentType: 'application/json',
            body: JSON.stringify(body),
          });
        });
      },

      async put(urlPattern: string, body: unknown, opts: MockOptions = {}) {
        await page.route(`**${urlPattern}**`, async route => {
          if (route.request().method() !== 'PUT') { await route.continue(); return; }
          await route.fulfill({
            status: opts.status ?? 200,
            contentType: 'application/json',
            body: JSON.stringify(body),
          });
        });
      },

      async delete(urlPattern: string, opts: MockOptions = {}) {
        await page.route(`**${urlPattern}**`, async route => {
          if (route.request().method() !== 'DELETE') { await route.continue(); return; }
          await route.fulfill({ status: opts.status ?? 204 });
        });
      },

      async error(urlPattern: string, status: number, message = 'Error') {
        await page.route(`**${urlPattern}**`, route =>
          route.fulfill({
            status,
            contentType: 'application/json',
            body: JSON.stringify({ message }),
          })
        );
      },

      async networkError(urlPattern: string) {
        await page.route(`**${urlPattern}**`, route => route.abort('failed'));
      },

      async passThrough(urlPattern: string) {
        await page.route(`**${urlPattern}**`, route => route.continue());
      },
    };

    await use(mock);
  },
});

export { expect } from '@playwright/test';
```

## Fixture Composition (Combined Export)

```typescript
// apps/web/e2e/fixtures/index.ts
import { mergeTests, mergeExpects } from '@playwright/test';
import { test as authTest, expect as authExpect } from './auth.fixture';
import { test as mockTest, expect as mockExpect } from './api-mock.fixture';

// Merge all fixtures into a single test function
export const test = mergeTests(authTest, mockTest);
export const expect = mergeExpects(authExpect, mockExpect);
```

## Global Setup — Auth State Generation

```typescript
// apps/web/e2e/global-setup.ts
import { chromium, type FullConfig } from '@playwright/test';
import path from 'path';
import fs from 'fs';

const ROLES = ['superadmin', 'admin', 'viewer', 'readonly'] as const;

export default async function globalSetup(config: FullConfig) {
  const authDir = path.join(process.cwd(), 'apps/web/e2e/.auth');
  fs.mkdirSync(authDir, { recursive: true });

  for (const role of ROLES) {
    const stateFile = path.join(authDir, `${role}.json`);

    // Skip if state file exists and is recent (< 1 hour old)
    if (fs.existsSync(stateFile)) {
      const stat = fs.statSync(stateFile);
      const ageMs = Date.now() - stat.mtimeMs;
      if (ageMs < 3_600_000) continue;
    }

    const browser = await chromium.launch();
    const page = await browser.newPage();

    // Navigate to mock login and authenticate as this role
    await page.goto('http://localhost:3000/login-mock');
    await page.getByText(roleToDisplayName(role), { exact: false }).click();
    await page.waitForURL('**/secure', { timeout: 15_000 });

    // Save auth state
    await page.context().storageState({ path: stateFile });
    await browser.close();
  }
}

function roleToDisplayName(role: string): string {
  const names: Record<string, string> = {
    superadmin: 'Superadmin',
    admin:      'Admin',
    viewer:     'Viewer',
    readonly:   'Read Only',
  };
  return names[role] ?? role;
}
```

Add to `playwright.config.ts`:
```typescript
export default defineConfig({
  globalSetup: './e2e/global-setup.ts',
  // …
});
```

## Usage in Tests

```typescript
// ✅ Always import from fixtures, never from @playwright/test directly
import { test, expect } from '../../fixtures';
import { UsersListPage } from '../../pages/users-list.page';
import { faker } from '@faker-js/faker';

test.describe('Users', () => {
  // Use authenticatedPage for full access tests
  test('admin can create user @smoke', async ({ authenticatedPage, mockApi }) => {
    const page = authenticatedPage;
    const listPage = new UsersListPage(page);

    await mockApi.post('/api/users', { id: '123', name: 'John' });
    await listPage.goto();
    await listPage.clickNewUser();
    // …
  });

  // Use pageAs for role-based tests
  test('viewer cannot access create button', async ({ pageAs }) => {
    const page = await pageAs('viewer');
    const listPage = new UsersListPage(page);
    await listPage.goto();
    await expect(listPage.newUserBtn).not.toBeVisible();
  });
});
```

## Token Optimization

**Load when** when designing Playwright auth fixtures or storage state. **Load only**: this pattern + the standard that owns its domain. **Unload after** the change is committed and verified — patterns are reference material, not session-resident.
