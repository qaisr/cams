# Playwright E2E Standards — NextJS / TypeScript / CSR / PingID

> **Version:** 2.0 | Enforced by `/playwright-ui-audit`

## 1. SELECTOR HIERARCHY (MANDATORY)

```
Priority 1: page.getByTestId('…')        — interactive/feature elements
Priority 2: page.getByRole('…', {name})  — semantic elements
Priority 3: page.getByLabel('…')         — form inputs
Priority 4: page.getByText(/…/i)         — static read-only content ONLY
NEVER:      .locator('.class')           — CSS classes
NEVER:      .locator('//xpath')          — XPath
NEVER:      .locator('tag')             — tag-only
NEVER:      .locator('#id')             — IDs (unstable)
NEVER:      .nth(0)                     — positional (unless in Page Object with comment)
```

### data-testid Naming Convention

```
{feature}-{element}-{variant?}

✅ user-submit-btn
✅ invoice-status-badge
✅ pipeline-stage-card
✅ modal-confirm-btn
✅ search-input
✅ filter-status-dropdown
✅ pagination-next-btn
✅ empty-state-message
✅ loading-spinner
✅ error-alert
✅ success-toast

❌ btn1
❌ submitButton
❌ theModal
❌ div1
```

Dynamic list items:
```tsx
// Component
data-testid={`user-row-${user.id}`}
data-testid={`invoice-amount-${invoice.id}`}

// Test
page.getByTestId(`user-row-${userId}`)
// OR for unknown IDs:
page.getByRole('row', { name: userName })
```

## 3. PAGE OBJECT MODEL (CANONICAL PATTERN)

```typescript
// apps/web/e2e/pages/users-list.page.ts
import { type Page, type Locator, expect } from '@playwright/test';

export class UsersListPage {
  // ── Locators (ALL readonly, defined in constructor) ──────────────
  readonly heading: Locator;
  readonly table: Locator;
  readonly newUserBtn: Locator;
  readonly searchInput: Locator;
  readonly paginationNext: Locator;
  readonly paginationPrev: Locator;
  readonly loadingSpinner: Locator;
  readonly emptyState: Locator;
  readonly successAlert: Locator;
  readonly errorAlert: Locator;

  constructor(readonly page: Page) {
    this.heading       = page.getByRole('heading', { name: /users/i });
    this.table         = page.getByRole('table', { name: /users/i });
    this.newUserBtn    = page.getByTestId('new-user-btn');
    this.searchInput   = page.getByTestId('search-input');
    this.paginationNext = page.getByTestId('pagination-next-btn');
    this.paginationPrev = page.getByTestId('pagination-prev-btn');
    this.loadingSpinner = page.getByTestId('loading-spinner');
    this.emptyState    = page.getByTestId('empty-state-message');
    this.successAlert  = page.getByRole('alert').filter({ hasText: /success/i });
    this.errorAlert    = page.getByRole('alert').filter({ hasText: /error/i });
  }

  // ── Navigation ───────────────────────────────────────────────────
  async goto(): Promise<void> {
    await this.page.goto('/users');
    await expect(this.heading).toBeVisible({ timeout: 10_000 });
  }

  // ── Actions (NO assertions inside action methods) ─────────────────
  async clickNewUser(): Promise<void> {
    await this.newUserBtn.click();
    await this.page.waitForURL('**/users/new');
  }

  async search(term: string): Promise<void> {
    await this.searchInput.fill(term);
    await this.searchInput.press('Enter');
    await this.page.waitForLoadState('networkidle');
  }

  async goToNextPage(): Promise<void> {
    await this.paginationNext.click();
    await this.page.waitForLoadState('networkidle');
  }

  // ── Row helpers ──────────────────────────────────────────────────
  getRow(name: string): Locator {
    return this.table.getByRole('row', { name: new RegExp(name, 'i') });
  }

  getRowActionBtn(name: string, action: 'edit' | 'delete' | 'view'): Locator {
    return this.getRow(name).getByTestId(`user-${action}-btn`);
  }
}
```

**Rules:**
- ✅ Locators are `readonly` properties set in the constructor
- ✅ Action methods use `async/await` and return `Promise<void>`
- ✅ Action methods do NOT contain `expect()` calls
- ✅ `goto()` waits for page to be ready before returning
- ✅ Row/dynamic getters are methods (not readonly properties)
- ❌ Do NOT use `page.locator()` directly in test files
- ❌ Do NOT define locators as strings
- ❌ Do NOT put assertions inside Page Object methods

```typescript
import { test, expect } from '../fixtures';  // Always import from fixtures/index.ts
import { faker } from '@faker-js/faker';

test.describe('Users — List', () => {
  test.beforeEach(async ({ authenticatedPage }) => {
    // Use fixture for auth, not manual login in every test
  });

  test('displays user list with column headers @smoke', async ({ page }) => {
    // Test body
  });

  test('shows empty state when no users exist', async ({ page, mockApi }) => {
    await mockApi.get('/api/users', { users: [], total: 0 });
    // Test body
  });

```

### Test Naming Convention

```
'[outcome] when [condition]'     ← preferred format
'[action] [result]'              ← acceptable

✅ 'displays validation error when name is empty'
✅ 'redirects to login when session expires'
✅ 'shows success toast after user is created'
✅ 'disables submit button while request is in flight'

❌ 'test user creation'
❌ 'check form'
❌ 'test 1'
❌ 'it works'
```

### Test Tags

```typescript
test('critical path @smoke', …)        // Run in CI pre-deploy
test('full coverage @regression', …)   // Run nightly
test('a11y check @accessibility', …)   // Run in accessibility CI job
test('slow test @slow', …)             // Excluded from fast CI
```

Run by tag:
```bash
npx playwright test --grep @smoke
npx playwright test --grep-invert @slow
```

## 5. FIXTURES (MANDATORY)

Always use fixtures from `apps/web/e2e/fixtures/index.ts`.
Never authenticate manually inside individual tests.

```typescript
// fixtures/index.ts — single export point
export { test, expect } from './auth.fixture';
export type { AuthFixtures } from './auth.fixture';
```

Available fixtures:
- `authenticatedPage` — page with Superadmin PingID session
- `authenticatedPageAs(role)` — page authenticated as specific role
- `mockApi` — pre-configured route interceptor
- `testData` — faker-based data factory

## 6. API MOCKING STRATEGY

```typescript
// Always mock APIs in unit/feature tests
// Only use real APIs in smoke/integration tests against staging

// ── Mock a GET endpoint ──────────────────────────────────────────
await page.route('**/api/users**', route => route.fulfill({
  status: 200,
  contentType: 'application/json',
  body: JSON.stringify({ users: [mockUser], total: 1, page: 0, limit: 20 }),
}));

// ── Mock an error state ──────────────────────────────────────────
await page.route('**/api/users', route => route.fulfill({
  status: 500,
  contentType: 'application/json',
  body: JSON.stringify({ message: 'Internal Server Error' }),
}));

// ── Assert API was called ────────────────────────────────────────
const apiCall = page.waitForRequest(req =>
  req.url().includes('/api/users') && req.method() === 'POST'
);
await page.getByTestId('user-submit-btn').click();
const request = await apiCall;
expect(request.postDataJSON()).toMatchObject({ name: 'John' });

// ── Mock ONCE then pass through ──────────────────────────────────
await page.route('**/api/users', async route => {
  if (route.request().method() === 'GET') {
    await route.fulfill({ status: 200, body: JSON.stringify(mockData) });
  } else {
    await route.continue();
  }
});
```

## 7. WAITING STRATEGY (CSR-SPECIFIC)

```typescript
// ✅ After navigation — always wait for meaningful content
await page.goto('/users');
await expect(page.getByTestId('users-page-heading')).toBeVisible();

// ✅ After form submit — wait for API response OR success indicator
const responsePromise = page.waitForResponse(
  resp => resp.url().includes('/api/users') && resp.status() === 201
);
await page.getByTestId('user-submit-btn').click();
await responsePromise;
await expect(page.getByTestId('success-toast')).toBeVisible();

// ✅ After client-side navigation (no page reload)
await page.getByTestId('user-row-123').click();
await page.waitForURL('**/users/123');
await expect(page.getByTestId('user-detail-heading')).toBeVisible();

// ✅ For loading states
await expect(page.getByTestId('loading-spinner')).toBeVisible();
await expect(page.getByTestId('loading-spinner')).toBeHidden({ timeout: 10_000 });

// ❌ NEVER
await page.waitForTimeout(2000);
```

## 8. ZOD VALIDATION TEST PATTERN

When a feature has Zod schemas in `packages/validation/src/`, generate
one test per validation rule:

```typescript
import { invoiceSchema } from '@repo/validation';
import { faker } from '@faker-js/faker';

// Auto-derive test cases from Zod schema
test.describe('Invoice Form — Validation', () => {
  test('shows error when amount is negative', async ({ page }) => {
    await formPage.goto();
    await formPage.fill({ amount: -1 });
    await formPage.save();
    await expect.soft(formPage.getFieldError('Amount'))
      .toHaveText(/must be positive/i);
    await expect(page).toHaveURL(/\/invoices\/new$/); // stays on form
  });

  test('shows error when due date is in the past', async ({ page }) => {
    await formPage.goto();
    await formPage.fill({ dueDate: '2020-01-01' });
    await formPage.save();
    await expect.soft(formPage.getFieldError('Due Date'))
      .toHaveText(/must be a future date/i);
  });
});
```

## 9. ACCESSIBILITY TEST PATTERN

```typescript
test.describe('Accessibility @accessibility', () => {
  test('page has single h1', async ({ page }) => {
    await listPage.goto();
    await expect(page.getByRole('heading', { level: 1 })).toHaveCount(1);
  });

  test('table has aria-label', async ({ page }) => {
    await listPage.goto();
    await expect(listPage.table).toHaveAttribute('aria-label');
  });

  test('form inputs have associated labels', async ({ page }) => {
    await formPage.goto();
    // All required inputs should be reachable via getByLabel
    await expect(page.getByLabel(/name/i)).toBeVisible();
    await expect(page.getByLabel(/email/i)).toBeVisible();
  });

  test('modal traps focus when open', async ({ page }) => {
    await listPage.goto();
    await page.getByTestId('delete-btn').click();
    const modal = page.getByRole('dialog');
    await expect(modal).toBeVisible();
    // Tab should stay within modal
    await page.keyboard.press('Tab');
    await expect(modal.getByRole('button').first()).toBeFocused();
  });

  test('error messages are announced via aria-live', async ({ page }) => {
    await formPage.goto();
    await formPage.save();
    const errorRegion = page.getByRole('alert');
    await expect(errorRegion).toBeVisible();
    await expect(errorRegion).toHaveAttribute('aria-live');
  });

  test('keyboard navigation reaches all interactive elements', async ({ page }) => {
    await listPage.goto();
    // Tab through all focusable elements and verify logical order
    await page.keyboard.press('Tab'); // Skip to main content link
    await page.keyboard.press('Tab'); // New user button
    await expect(listPage.newUserBtn).toBeFocused();
    await page.keyboard.press('Tab'); // Search input
    await expect(listPage.searchInput).toBeFocused();
  });
});
```

## 10. ANTI-PATTERNS (ENFORCED BY /playwright-ui-audit)

```typescript
// ❌ CRITICAL — arbitrary waits
await page.waitForTimeout(2000);

// ❌ CRITICAL — CSS selectors
await page.locator('.action-btn').click();
page.locator('#submit');

// ❌ CRITICAL — non-retrying assertions
const text = await page.locator('h1').textContent();
expect(text).toBe('Dashboard');
// ✅ Use: await expect(page.locator('h1')).toHaveText('Dashboard');

// ❌ CRITICAL — assertions inside Page Object methods
class MyPage {
  async clickSave() {
    await this.saveBtn.click();
    await expect(this.successAlert).toBeVisible(); // ← WRONG
  }
}

// ❌ WARNING — hard-coded test data
test('creates user', async ({ page }) => {
  await formPage.fill({ name: 'John Smith', email: 'john@test.com' });
});
// ✅ Use faker:
const name = faker.person.fullName();
const email = faker.internet.email();

// ❌ WARNING — dependent tests without serial
test.describe('checkout', () => {
  test('adds item to cart', …);
  test('checks out', …);     // ← depends on previous test state
});
// ✅ Use:
test.describe.serial('checkout flow', () => { … });
```
